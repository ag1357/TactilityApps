cmake_minimum_required(VERSION 3.20)
project(anaphorum_conversation_test C)
if(NOT IS_ABSOLUTE "${LVGL_SOURCE}" OR NOT EXISTS "${LVGL_SOURCE}/lvgl.h")
    message(FATAL_ERROR "Pass -DLVGL_SOURCE=/absolute/path/to/LVGL-9.3-source")
endif()
set(LV_BUILD_CONF_PATH "${CMAKE_CURRENT_SOURCE_DIR}/lv_conf.h" CACHE PATH "" FORCE)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG_INTERNAL OFF CACHE BOOL "" FORCE)
add_subdirectory("${LVGL_SOURCE}" lvgl)
add_executable(conversation_lvgl test.c ../../core/conversation_layout.c)
target_include_directories(conversation_lvgl PRIVATE ../../core ../../main)
target_compile_options(conversation_lvgl PRIVATE -Wall -Wextra -Werror)
target_link_libraries(conversation_lvgl PRIVATE lvgl::lvgl m)
enable_testing()
add_test(NAME conversation_lvgl COMMAND conversation_lvgl)
